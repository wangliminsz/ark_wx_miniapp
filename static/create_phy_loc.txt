2. JS 逻辑（核心！直接复制）
pages/xxx/xxx.js
javascript
运行
const config = require('../../../config.js'); // 你的配置文件路径

Page({
  data: {
    loc_code: '',
    loc_name: '',
    remark: '',
    loading: false,
  },

  // 输入绑定
  onLocCodeInput(e) {
    this.setData({ loc_code: e.detail.value });
  },
  onLocNameInput(e) {
    this.setData({ loc_name: e.detail.value });
  },
  onRemarkInput(e) {
    this.setData({ remark: e.detail.value });
  },

  // =====================================
  // ✅ 创建物理库位（对接你FastAPI接口）
  // =====================================
  createPhysicalLocation() {
    const { loc_code, loc_name, remark } = this.data;

    // 简单校验
    if (!loc_code || !loc_name) {
      wx.showToast({ title: '编码和名称不能为空', icon: 'none' });
      return;
    }

    this.setData({ loading: true });
    wx.showLoading({ title: '创建中...' });

    // 获取token
    const token = wx.getStorageSync('odoo_user_erp_token');
    if (!token) {
      wx.showToast({ title: '请先登录', icon: 'none' });
      this.setData({ loading: false });
      return;
    }

    // ✅ 调用 FastAPI
    wx.request({
      url: config.fastapiUrl + '/physical/locations',
      method: 'POST',
      header: {
        'Authorization': 'Bearer ' + token,
        'Content-Type': 'application/json',
      },
      data: {
        loc_code: loc_code,
        loc_name: loc_name,
        remark: remark || ''
      },
      success: (res) => {
        console.log('创建成功:', res.data);
        
        if (res.statusCode === 200) {
          wx.showToast({ title: '创建成功 ✅', icon: 'success' });
          
          // 创建成功后清空表单
          this.setData({
            loc_code: '',
            loc_name: '',
            remark: ''
          });
        } else {
          wx.showToast({
            title: res.data.detail || '创建失败',
            icon: 'none'
          });
        }
      },
      fail: (err) => {
        console.error('请求失败', err);
        wx.showToast({ title: '网络错误', icon: 'none' });
      },
      complete: () => {
        this.setData({ loading: false });
        wx.hideLoading();
      }
    });
  },
});
