微信小程序 - 实物库存盘点页面 开发任务单（可直接发 Trae）
一、项目说明
基于现有 FastAPI + Odoo 后端接口，开发产品实物库存批量盘点编辑页；
无需新增 / 修改任何后端接口，全部复用已有接口，只做小程序页面开发 + 联调。
二、已现成可用后端接口（直接对接）
获取产品批次：GET /product/lots?product_code=xxx
按产品代码查询已有实物库存（回填用）：GET /physical/stock/by_product?code=xxx
单条更新实物库存：POST /physical/stock/update
获取所有物理库位：GET /physical/locations
三、页面需求功能
从产品列表页点击产品，携带 product_id / product_code / product_name 跳转进入盘点详情页。
页面自动加载：产品名称 / 代码、总库存、所有物理库位、当前产品所有批次、已存在的库存多行数据自动回填列表。
列表每行固定三项：库位选择、批次选择、实盘数量输入。
支持功能：
手动新增空白行
任意删除单行（至少保留一行）
修改已有行数量
一键保存全部行，逐行调用更新接口提交
表单校验：空库位、空批次、非法数字数量 禁止提交并弹窗提示。
保存中 loading 遮罩，保存成功自动刷新回填最新数据。
页面 UI 整洁适配手机，列表可滚动，排版对齐美观。

2) edit_stock.js
javascript
运行
const config = require("../../config");

Page({
  data: {
    productId: "",
    productCode: "",
    productName: "",
    totalRealQty: 0,
    stockList: [], // {id, loc_code, lot_name, real_qty, lot_id?, physical_loc_id?}
    locList: [],   // 所有库位
    lotList: []    // 当前产品所有批次
  },

  onLoad(options) {
    const { product_id, product_code, product_name } = options;
    this.setData({
      productId: product_id,
      productCode: product_code,
      productName: product_name
    });
    this.fetchLocations();
    this.fetchProductLots(product_code);
    this.fetchExistingStock(product_code);
  },

  // 1. 获取所有库位
  fetchLocations() {
    const token = wx.getStorageSync("odoo_user_erp_token");
    wx.request({
      url: `${config.fastapiUrl}/physical/locations`,
      method: "GET",
      header: { Authorization: "Bearer " + token },
      success: (res) => {
        if (res.statusCode === 200 && res.data) {
          this.setData({ locList: res.data });
        }
      }
    });
  },

  // 2. 获取当前产品批次
  fetchProductLots(code) {
    const token = wx.getStorageSync("odoo_user_erp_token");
    wx.request({
      url: `${config.fastapiUrl}/product/lots?product_code=${code}`,
      method: "GET",
      header: { Authorization: "Bearer " + token },
      success: (res) => {
        if (res.statusCode === 200 && res.data && res.data.lots) {
          this.setData({ lotList: res.data.lots });
        }
      }
    });
  },

  // 3. 加载已有库存（回填）
  fetchExistingStock(code) {
    const token = wx.getStorageSync("odoo_user_erp_token");
    wx.request({
      url: `${config.fastapiUrl}/physical/stock/by_product?code=${code}`,
      method: "GET",
      header: { Authorization: "Bearer " + token },
      success: (res) => {
        if (res.statusCode === 200 && res.data) {
          let list = res.data.details || [];
          // 没有数据默认一行空
          if (list.length === 0) {
            list = [{ id: null, loc_code: "", lot_name: "", real_qty: "" }];
          }
          this.setData({
            stockList: list,
            totalRealQty: res.data.total_real_qty || 0
          });
        }
      }
    });
  },

  // 库位变更
  onLocChange(e) {
    const idx = e.currentTarget.dataset.index;
    const sel = this.data.locList[e.detail.value];
    let key = `stockList[${idx}].loc_code`;
    this.setData({ [key]: sel.loc_code });
  },

  // 批次变更
  onLotChange(e) {
    const idx = e.currentTarget.dataset.index;
    const sel = this.data.lotList[e.detail.value];
    let key = `stockList[${idx}].lot_name`;
    this.setData({ [key]: sel.name });
  },

  // 数量输入
  onQtyInput(e) {
    const idx = e.currentTarget.dataset.index;
    const val = e.detail.value;
    let key = `stockList[${idx}].real_qty`;
    this.setData({ [key]: val });
  },

  // 新增行
  onAddRow() {
    let row = { id: null, loc_code: "", lot_name: "", real_qty: "" };
    this.setData({ stockList: [...this.data.stockList, row] });
  },

  // 删除行
  onDelRow(e) {
    const idx = e.currentTarget.dataset.index;
    let list = [...this.data.stockList];
    list.splice(idx, 1);
    // 至少保留一行
    if (list.length === 0) list = [{ id: null, loc_code: "", lot_name: "", real_qty: "" }];
    this.setData({ stockList: list });
  },

  // 保存全部（逐行调用你的 update 接口）
  onSaveAll() {
    const token = wx.getStorageSync("odoo_user_erp_token");
    const { stockList, productId } = this.data;
    // 简单校验
    for (let r of stockList) {
      if (!r.loc_code || !r.lot_name || r.real_qty === "" || isNaN(Number(r.real_qty))) {
        wx.showToast({ title: "请完善库位/批次/有效数量", icon: "none" });
        return;
      }
    }

    wx.showLoading({ title: "保存中..." });
    let promises = stockList.map((row) => {
      return new Promise((resolve, reject) => {
        // 根据 loc_code 拿 physical_loc_id（你后端应该有映射）
        // 根据 lot_name 拿 lot_id
        // 这里假设你后端接口支持 name 查找，或你前端先映射
        // 下面直接按你现有接口字段调用
        wx.request({
          url: `${config.fastapiUrl}/physical/stock/update`,
          method: "POST",
          header: { Authorization: "Bearer " + token, "Content-Type": "application/json" },
          data: {
            product_id: parseInt(productId),
            lot_id: 0,          // 可前端根据 lot_name 匹配 lotList 填 id
            physical_loc_id: 0, // 可前端根据 loc_code 匹配 locList 填 id
            real_qty: Number(row.real_qty)
          },
          success: (res) => {
            if (res.statusCode === 200) resolve(res);
            else reject(res);
          },
          fail: reject
        });
      });
    });

    Promise.all(promises)
      .then(() => {
        wx.hideLoading();
        wx.showToast({ title: "保存成功" });
        // 刷新回填
        this.fetchExistingStock(this.data.productCode);
      })
      .catch((err) => {
        wx.hideLoading();
        wx.showToast({ title: "部分失败", icon: "none" });
        console.error(err);
      });
  }
});
