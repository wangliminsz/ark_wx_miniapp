2. JS 逻辑（核心代码，100% 可用）
javascript
运行
const config = require('../../../config.js');

Page({
  data: {
    locations: [],
    searchKey: '',
    loading: false,

    // 修改弹窗
    showEditModal: false,
    editData: {},

    // 选择弹窗
    showPickerModal: false,
    selectedLocation: null,
  },

  onLoad() {
    this.fetchLocations();
  },

  // =====================================
  // 1. 获取库位列表
  // =====================================
  fetchLocations() {
    const token = wx.getStorageSync('odoo_user_erp_token');
    if (!token) return;

    this.setData({ loading: true });

    wx.request({
      url: config.fastapiUrl + '/physical/locations',
      method: 'GET',
      header: {
        'Authorization': 'Bearer ' + token
      },
      success: (res) => {
        if (res.statusCode === 200) {
          this.setData({ locations: res.data.records });
        }
      },
      complete: () => {
        this.setData({ loading: false });
      }
    });
  },

  // =====================================
  // 2. 删除库位
  // =====================================
  deleteLocation(e) {
    const code = e.currentTarget.dataset.code;
    const token = wx.getStorageSync('odoo_user_erp_token');

    wx.showModal({
      title: '确认删除',
      content: '删除后无法恢复，且有库存无法删除',
      success: (res) => {
        if (res.confirm) {
          wx.request({
            url: config.fastapiUrl + `/physical/locations/${code}`,
            method: 'DELETE',
            header: { 'Authorization': 'Bearer ' + token },
            success: (res) => {
              if (res.statusCode === 200) {
                wx.showToast({ title: '删除成功' });
                this.fetchLocations();
              } else {
                wx.showToast({ title: res.data.detail || '删除失败', icon: 'none' });
              }
            }
          });
        }
      }
    });
  },

  // =====================================
  // 3. 修改弹窗相关
  // =====================================
  openEditModal(e) {
    this.setData({
      showEditModal: true,
      editData: e.currentTarget.dataset.item
    });
  },
  closeEditModal() {
    this.setData({ showEditModal: false });
  },
  onEditNameInput(e) {
    this.setData({ 'editData.loc_name': e.detail.value });
  },
  onEditRemarkInput(e) {
    this.setData({ 'editData.remark': e.detail.value });
  },

  // 提交修改
  submitEdit() {
    const { editData } = this.data;
    const token = wx.getStorageSync('odoo_user_erp_token');

    wx.request({
      url: config.fastapiUrl + `/physical/locations/${editData.loc_code}`,
      method: 'PUT',
      header: {
        'Authorization': 'Bearer ' + token,
        'Content-Type': 'application/json'
      },
      data: {
        loc_name: editData.loc_name,
        remark: editData.remark
      },
      success: (res) => {
        if (res.statusCode === 200) {
          wx.showToast({ title: '保存成功' });
          this.closeEditModal();
          this.fetchLocations();
        }
      }
    });
  },

  // =====================================
  // 4. 库位选择弹窗
  // =====================================
  openLocationPicker() {
    this.setData({ showPickerModal: true });
  },
  closePicker() {
    this.setData({ showPickerModal: false });
  },
  selectLocation(e) {
    const item = e.currentTarget.dataset.item;
    this.setData({
      selectedLocation: item,
      showPickerModal: false
    });
    wx.showToast({ title: `已选择: ${item.loc_code}` });

    // 选择后你可以在这里做其他逻辑，比如回填表单
  },

  // =====================================
  // 搜索
  // =====================================
  onSearchInput(e) {
    this.setData({ searchKey: e.detail.value });
  }
});
